% Common axes for the two-good price diagrams.
\newcommand{\AuctionPriceAxes}{%
    \draw[-stealth,thick] (0,0) -- (64,0) node[right] {$p_1$};
    \draw[-stealth,thick] (0,0) -- (0,64) node[above] {$p_2$};
    \foreach \x in {10,20,...,60}
        \draw (\x,0.8) -- (\x,-0.8) node[below] {\scriptsize \x};
    \foreach \y in {10,20,...,60}
        \draw (0.8,\y) -- (-0.8,\y) node[left] {\scriptsize \y};
}

% Larger axes for diagrams involving Rainy's updated value of 80.
\newcommand{\AuctionPriceAxesWide}{%
    \draw[-stealth,thick] (0,0) -- (94,0) node[right] {$p_1$};
    \draw[-stealth,thick] (0,0) -- (0,94) node[above] {$p_2$};
    \foreach \x in {10,20,...,90}
        \draw (\x,0.8) -- (\x,-0.8) node[below] {\scriptsize \x};
    \foreach \y in {10,20,...,90}
        \draw (0.8,\y) -- (-0.8,\y) node[left] {\scriptsize \y};
}

% The Newton polygon of
% (x \oplus 0)(y \oplus 0)(xy \oplus 0), together with its two
% Minkowski-sum decompositions from the final example in the DRP notes.
\newcommand{\NewtonPolygonShape}{%
    \filldraw[fill=tropicalLift!20,draw=tropicalLift!85!black,
        thick,line join=round]
        (0,0) -- (1,0) -- (2,1) -- (2,2) -- (1,2) -- (0,1) -- cycle;
}

\newcommand{\NewtonPolygonAxesAndPoints}{%
    \draw[-stealth] (-0.22,0) -- (2.48,0) node[right] {$i$};
    \draw[-stealth] (0,-0.22) -- (0,2.48) node[above] {$j$};
    \foreach \x in {1,2}
        \draw (\x,0.045) -- (\x,-0.045) node[below=2pt] {\scriptsize \x};
    \foreach \y in {1,2}
        \draw (0.045,\y) -- (-0.045,\y) node[left=2pt] {\scriptsize \y};
    \node[below left=1pt] at (0,0) {\scriptsize $0$};
    \foreach \point in {(0,0),(1,0),(2,1),(2,2),(1,2),(0,1)}
        \fill \point circle[radius=1.35pt];
}
